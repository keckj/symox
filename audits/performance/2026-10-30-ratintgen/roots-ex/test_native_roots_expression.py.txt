@pytest.mark.parametrize('text,expected_dict,expected_list', CASES)
def test_roots_match_sympy(text, expected_dict, expected_list):
    p = sx.sympify(text, locals=NAMES)
    assert str(sx.roots(p, x)) == expected_dict
    assert str(sx.roots(p, x, multiple=True)) == expected_list
